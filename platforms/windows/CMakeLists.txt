cmake_minimum_required(VERSION 3.20)
project(chengyin_windows VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This preview requires a Windows x64 toolchain")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CHENGYIN_DAILY_DICTIONARY "${CMAKE_CURRENT_SOURCE_DIR}/../../data/daily.mswydict")
# The release version header comes from version.json, the single audited source
# (review R13). Build.ps1 regenerates it before configuring; a stale or hand-edited
# header fails loudly here rather than silently shipping the wrong tag.
find_package(Python3 COMPONENTS Interpreter QUIET)
if(NOT Python3_Interpreter_FOUND)
  message(FATAL_ERROR "Python 3 is required to verify the release version source")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/version.py" --check
                RESULT_VARIABLE CHENGYIN_VERSION_STATUS OUTPUT_VARIABLE CHENGYIN_VERSION_TEXT
                ERROR_VARIABLE CHENGYIN_VERSION_ERROR)
if(NOT CHENGYIN_VERSION_STATUS EQUAL 0)
  message(FATAL_ERROR "Release version source is not in sync: ${CHENGYIN_VERSION_TEXT}${CHENGYIN_VERSION_ERROR}")
endif()
configure_file(daily_data.rc.in daily_data.rc @ONLY)
configure_file(app_manifest.rc.in app_manifest.rc @ONLY)
configure_file(icon.rc.in icon.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" PROPERTIES OBJECT_DEPENDS "${CHENGYIN_DAILY_DICTIONARY}")
set(CHENGYIN_CORE_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/x86_64-pc-windows-msvc/release/chengyin_ime.lib" CACHE FILEPATH "Rust target static library")
if(NOT EXISTS "${CHENGYIN_CORE_LIBRARY}")
  message(FATAL_ERROR "Build chengyin-ffi for the matching Windows target first: ${CHENGYIN_CORE_LIBRARY}")
endif()
add_library(chengyin_core STATIC IMPORTED)
set_target_properties(chengyin_core PROPERTIES IMPORTED_LOCATION "${CHENGYIN_CORE_LIBRARY}")
add_library(windows_options INTERFACE)
target_include_directories(windows_options INTERFACE ../../include)
target_link_libraries(windows_options INTERFACE advapi32 winhttp bcrypt)
target_compile_definitions(windows_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  # Bounded legacy CRT calls are reviewed explicitly; keep all other /W4 warnings fatal.
  target_compile_definitions(windows_options INTERFACE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(windows_options INTERFACE /W4 /WX /utf-8 /permissive-)
else()
  target_compile_options(windows_options INTERFACE -Wall -Wextra -Werror -Wno-cast-function-type)
  target_link_options(windows_options INTERFACE -static -static-libgcc -static-libstdc++)
endif()
# Rust std's Windows static-link dependencies; no extra runtime DLL is packaged.
target_link_libraries(chengyin_core INTERFACE ws2_32 userenv bcrypt ntdll advapi32 synchronization dbghelp)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/skins/whale-girl.png" WHALE_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," WHALE_BYTES "${WHALE_HEX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/whale_data.cpp" "#include <cstddef>\nnamespace chengyin { extern const unsigned char whalePng[] = {${WHALE_BYTES}}; extern const size_t whalePngSize=sizeof(whalePng); }\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/skins/whale-girl.png")
add_library(windows_theme_art STATIC theme_art.cpp skin.cpp "${CMAKE_CURRENT_BINARY_DIR}/whale_data.cpp")
target_link_libraries(windows_theme_art PUBLIC windows_options gdiplus crypt32 ole32 msimg32)
add_library(chengyin_tsf SHARED dll.cpp service.cpp candidate.cpp mode_hint.cpp settings.cpp dictionary_source.cpp dictionary_library.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp ui_element.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" chengyin.def)
target_include_directories(chengyin_tsf PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(chengyin_tsf PRIVATE windows_options chengyin_core ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 comdlg32)
set_target_properties(chengyin_tsf PROPERTIES PREFIX "")
add_executable(chengyin_probe probe.cpp test_edits.cpp)
target_link_libraries(chengyin_probe PRIVATE windows_options ole32 oleaut32 uuid bcrypt advapi32)
add_executable(chengyin_settings WIN32 settings_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" settings.cpp dictionary_library.cpp preferences.cpp settings_ui.cpp update.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/app_manifest.rc")
target_link_libraries(chengyin_settings PRIVATE windows_options chengyin_core ole32 uuid user32 gdi32 shell32 comdlg32 comctl32 advapi32 imm32)
if(MSVC)
  # Our RC already embeds the DPI/common-controls manifest. Do not add a second one.
  target_link_options(chengyin_settings PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(chengyin_probe PRIVATE -municode)
  target_link_options(chengyin_settings PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(windows_skin_test test_skin.cpp preferences.cpp settings.cpp)
  target_link_libraries(windows_skin_test PRIVATE windows_options windows_theme_art chengyin_core shell32 comdlg32 ole32 uuid user32 gdi32)
  target_compile_definitions(windows_skin_test PRIVATE CHENGYIN_ISOLATED_CONFIG_NOTIFICATIONS)
  add_test(NAME windows_skin_test COMMAND windows_skin_test)
  # Deliberately faulty registration for installer rollback tests; never installed.
  add_library(chengyin_registration_failure SHARED dll.cpp service.cpp candidate.cpp mode_hint.cpp settings.cpp dictionary_source.cpp dictionary_library.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp ui_element.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" chengyin.def)
  target_include_directories(chengyin_registration_failure PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(chengyin_registration_failure PRIVATE CHENGYIN_TEST_REGISTRATION_FAILURE)
  target_link_libraries(chengyin_registration_failure PRIVATE windows_options chengyin_core ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 comdlg32)
  set_target_properties(chengyin_registration_failure PROPERTIES PREFIX "")
  add_executable(windows_key_test test_keymap.cpp candidate.cpp preferences.cpp settings.cpp)
  target_link_libraries(windows_key_test PRIVATE windows_options chengyin_core user32 gdi32 shell32 comdlg32 ole32 uuid)
  # Same service implementation, with user vocabulary I/O isolated from regressions.
  add_library(chengyin_tsf_fixture SHARED dll.cpp service.cpp candidate.cpp mode_hint.cpp settings.cpp dictionary_source.cpp dictionary_library.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp ui_element.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" chengyin.def)
  target_compile_definitions(chengyin_tsf_fixture PRIVATE CHENGYIN_FIXED_TEST_VOCABULARY)
  target_link_libraries(chengyin_tsf_fixture PRIVATE windows_options chengyin_core ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 comdlg32)
  set_target_properties(chengyin_tsf_fixture PROPERTIES PREFIX "")
  add_executable(windows_settings_test test_settings.cpp settings.cpp dictionary_source.cpp dictionary_library.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc")
  target_link_libraries(windows_settings_test PRIVATE windows_options chengyin_core shell32 comdlg32 ole32 uuid user32 gdi32 advapi32)
  add_executable(windows_ui_test test_ui.cpp settings_ui.cpp update.cpp settings.cpp dictionary_library.cpp preferences.cpp candidate.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/app_manifest.rc")
  target_compile_definitions(windows_ui_test PRIVATE CHENGYIN_ISOLATED_UI_TEST)
  add_dependencies(windows_ui_test chengyin_tsf_fixture)
  target_link_libraries(windows_ui_test PRIVATE windows_options chengyin_core shell32 comdlg32 ole32 uuid user32 gdi32 advapi32 imm32 comctl32)
  if(MSVC)
    target_link_options(windows_ui_test PRIVATE /MANIFEST:NO)
  endif()
  if(MINGW)
    target_link_options(windows_ui_test PRIVATE -municode)
  endif()
  add_executable(windows_live_test test_live.cpp configuration.cpp language_bar.cpp preferences.cpp settings.cpp)
  if(MSVC)
    target_compile_options(windows_live_test PRIVATE /Od)
  endif()
  target_link_libraries(windows_live_test PRIVATE windows_options chengyin_core shell32 comdlg32 ole32 oleaut32 uuid user32 gdi32 advapi32)
  add_test(NAME windows_live_test COMMAND windows_live_test)
  add_executable(windows_update_test test_update.cpp update.cpp preferences.cpp settings.cpp)
  target_link_libraries(windows_update_test PRIVATE windows_options chengyin_core shell32 comdlg32 ole32 uuid user32 gdi32)
  add_test(NAME windows_update_test COMMAND windows_update_test)
  add_executable(windows_mode_hint_test test_mode_hint.cpp mode_hint.cpp candidate.cpp preferences.cpp settings.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc")
  target_link_libraries(windows_mode_hint_test PRIVATE windows_options windows_theme_art chengyin_core shell32 comdlg32 ole32 uuid user32 gdi32 advapi32)
  add_test(NAME windows_mode_hint_test COMMAND windows_mode_hint_test)
  # Optional manual desktop driver, not part of the default build or CTest.
  add_executable(windows_desktop_test EXCLUDE_FROM_ALL test_desktop.cpp settings_ui.cpp update.cpp settings.cpp dictionary_library.cpp preferences.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/app_manifest.rc")
  target_compile_definitions(windows_desktop_test PRIVATE CHENGYIN_ISOLATED_UI_TEST)
  target_link_libraries(windows_desktop_test PRIVATE windows_options chengyin_core shell32 comdlg32 ole32 oleaut32 uuid user32 gdi32 advapi32 imm32 comctl32)
  if(MSVC)
    target_link_options(windows_desktop_test PRIVATE /MANIFEST:NO)
  endif()
  if(MINGW)
    target_link_options(windows_desktop_test PRIVATE -municode)
  endif()
  add_test(NAME windows_ui_test COMMAND windows_ui_test)
  add_test(NAME windows_settings_test COMMAND windows_settings_test)
  add_test(NAME windows_key_test COMMAND windows_key_test)
  add_test(NAME windows_com_probe COMMAND chengyin_probe $<TARGET_FILE:chengyin_tsf>)
  add_test(NAME windows_tsf_edits COMMAND chengyin_probe $<TARGET_FILE:chengyin_tsf_fixture> --edits)
endif()
foreach(theme_target IN ITEMS chengyin_tsf chengyin_settings chengyin_registration_failure windows_key_test chengyin_tsf_fixture windows_ui_test windows_desktop_test windows_settings_test windows_live_test windows_update_test)
  if(TARGET ${theme_target})
    target_link_libraries(${theme_target} PRIVATE windows_theme_art)
  endif()
endforeach()
foreach(fixture_target IN ITEMS chengyin_tsf_fixture windows_ui_test windows_key_test windows_settings_test windows_live_test windows_update_test windows_mode_hint_test windows_desktop_test)
  if(TARGET ${fixture_target})
    target_compile_definitions(${fixture_target} PRIVATE CHENGYIN_ISOLATED_CONFIG_NOTIFICATIONS)
  endif()
endforeach()
install(TARGETS chengyin_tsf chengyin_probe chengyin_settings RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES README.md THIRD_PARTY.md CHANGELOG.md DESTINATION .)
install(FILES ../../LICENSE DESTINATION .)
