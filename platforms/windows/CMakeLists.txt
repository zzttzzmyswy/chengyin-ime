cmake_minimum_required(VERSION 3.20)
project(myswy_windows VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This preview requires a Windows x64 toolchain")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MYSWY_DAILY_DICTIONARY "${CMAKE_CURRENT_SOURCE_DIR}/../../data/daily.mswydict")
configure_file(daily_data.rc.in daily_data.rc @ONLY)
configure_file(app_manifest.rc.in app_manifest.rc @ONLY)
configure_file(icon.rc.in icon.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" PROPERTIES OBJECT_DEPENDS "${MYSWY_DAILY_DICTIONARY}")
set(MYSWY_CORE_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/x86_64-pc-windows-msvc/release/myswy_ime.lib" CACHE FILEPATH "Rust target static library")
if(NOT EXISTS "${MYSWY_CORE_LIBRARY}")
  message(FATAL_ERROR "Build myswy-ffi for the matching Windows target first: ${MYSWY_CORE_LIBRARY}")
endif()
add_library(myswy_core STATIC IMPORTED)
set_target_properties(myswy_core PROPERTIES IMPORTED_LOCATION "${MYSWY_CORE_LIBRARY}")
add_library(windows_options INTERFACE)
target_include_directories(windows_options INTERFACE ../../include)
target_compile_definitions(windows_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  # Bounded legacy CRT calls are reviewed explicitly; keep all other /W4 warnings fatal.
  target_compile_definitions(windows_options INTERFACE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(windows_options INTERFACE /W4 /WX /utf-8 /permissive-)
else()
  target_compile_options(windows_options INTERFACE -Wall -Wextra -Werror -Wno-cast-function-type)
  target_link_options(windows_options INTERFACE -static -static-libgcc -static-libstdc++)
endif()
# Rust std's Windows static-link dependencies; no extra runtime DLL is packaged.
target_link_libraries(myswy_core INTERFACE ws2_32 userenv bcrypt ntdll advapi32 synchronization dbghelp)
add_library(myswy_tsf SHARED dll.cpp service.cpp candidate.cpp settings.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp ui_element.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" myswy.def)
target_include_directories(myswy_tsf PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(myswy_tsf PRIVATE windows_options myswy_core ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 comdlg32)
set_target_properties(myswy_tsf PROPERTIES PREFIX "")
add_executable(myswy_probe probe.cpp test_edits.cpp)
target_link_libraries(myswy_probe PRIVATE windows_options ole32 oleaut32 uuid bcrypt advapi32)
add_executable(myswy_settings WIN32 settings_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" settings.cpp preferences.cpp settings_ui.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/app_manifest.rc")
target_link_libraries(myswy_settings PRIVATE windows_options myswy_core ole32 uuid user32 gdi32 shell32 comdlg32 comctl32 advapi32 imm32)
if(MINGW)
  target_link_options(myswy_probe PRIVATE -municode)
  target_link_options(myswy_settings PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  # Deliberately faulty registration for installer rollback tests; never installed.
  add_library(myswy_registration_failure SHARED dll.cpp service.cpp candidate.cpp settings.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp ui_element.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" myswy.def)
  target_include_directories(myswy_registration_failure PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_definitions(myswy_registration_failure PRIVATE MYSWY_TEST_REGISTRATION_FAILURE)
  target_link_libraries(myswy_registration_failure PRIVATE windows_options myswy_core ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 comdlg32)
  set_target_properties(myswy_registration_failure PROPERTIES PREFIX "")
  add_executable(windows_key_test test_keymap.cpp candidate.cpp preferences.cpp settings.cpp)
  target_link_libraries(windows_key_test PRIVATE windows_options myswy_core user32 gdi32 shell32 comdlg32 ole32 uuid)
  # Same service implementation, with user vocabulary I/O isolated from regressions.
  add_library(myswy_tsf_fixture SHARED dll.cpp service.cpp candidate.cpp settings.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp ui_element.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/icon.rc" myswy.def)
  target_compile_definitions(myswy_tsf_fixture PRIVATE MYSWY_FIXED_TEST_VOCABULARY)
  target_link_libraries(myswy_tsf_fixture PRIVATE windows_options myswy_core ole32 oleaut32 uuid user32 gdi32 advapi32 shell32 comdlg32)
  set_target_properties(myswy_tsf_fixture PROPERTIES PREFIX "")
  add_executable(windows_settings_test test_settings.cpp settings.cpp preferences.cpp learning_writer.cpp configuration.cpp language_bar.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc")
  target_link_libraries(windows_settings_test PRIVATE windows_options myswy_core shell32 comdlg32 ole32 uuid user32 gdi32 advapi32)
  add_executable(windows_ui_test test_ui.cpp settings_ui.cpp settings.cpp preferences.cpp candidate.cpp "${CMAKE_CURRENT_BINARY_DIR}/daily_data.rc" "${CMAKE_CURRENT_BINARY_DIR}/app_manifest.rc")
  target_link_libraries(windows_ui_test PRIVATE windows_options myswy_core shell32 comdlg32 ole32 uuid user32 gdi32 advapi32 imm32 comctl32)
  if(MINGW)
    target_link_options(windows_ui_test PRIVATE -municode)
  endif()
  add_executable(windows_live_test test_live.cpp configuration.cpp language_bar.cpp preferences.cpp settings.cpp)
  target_link_libraries(windows_live_test PRIVATE windows_options myswy_core shell32 comdlg32 ole32 oleaut32 uuid user32 gdi32 advapi32)
  add_test(NAME windows_live_test COMMAND windows_live_test)
  add_test(NAME windows_ui_test COMMAND windows_ui_test)
  add_test(NAME windows_settings_test COMMAND windows_settings_test)
  add_test(NAME windows_key_test COMMAND windows_key_test)
  add_test(NAME windows_com_probe COMMAND myswy_probe $<TARGET_FILE:myswy_tsf>)
  add_test(NAME windows_tsf_edits COMMAND myswy_probe $<TARGET_FILE:myswy_tsf_fixture> --edits)
endif()
install(TARGETS myswy_tsf myswy_probe myswy_settings RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES README.md THIRD_PARTY.md DESTINATION .)
install(FILES ../../LICENSE DESTINATION .)
