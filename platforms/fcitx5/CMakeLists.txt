cmake_minimum_required(VERSION 3.20)
project(myswy-fcitx5 VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
find_package(Fcitx5Core 5.1 REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(MYSWY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MYSWY_RUST_LIBRARY "${MYSWY_ROOT}/target/release/libmyswy_ime.a" CACHE FILEPATH "Rust static library")
if(NOT EXISTS "${MYSWY_RUST_LIBRARY}")
  message(FATAL_ERROR "Run cargo build --release -p myswy-ffi before CMake")
endif()
add_library(myswy-engine STATIC engine.cpp dictionary_loader.cpp)
set_target_properties(myswy-engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(myswy-engine PUBLIC cxx_std_17)
target_include_directories(myswy-engine PUBLIC "${MYSWY_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(myswy-engine PRIVATE -Wall -Wextra -Werror)
target_link_libraries(myswy-engine PUBLIC Fcitx5::Core Fcitx5::Config "${MYSWY_RUST_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS} m)

add_library(myswy MODULE factory.cpp)
set_target_properties(myswy PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
target_link_libraries(myswy PRIVATE myswy-engine)
target_link_options(myswy PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
set(MYSWY_ADDON_DIR "${FCITX_INSTALL_ADDONDIR}" CACHE PATH "Fcitx5 addon directory from distro SDK")
install(TARGETS myswy LIBRARY DESTINATION "${MYSWY_ADDON_DIR}")
install(FILES myswy-addon.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/addon" RENAME myswy.conf)
install(FILES myswy-inputmethod.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod" RENAME myswy.conf)
install(FILES "${MYSWY_ROOT}/LICENSE" DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/fcitx5-myswy" RENAME copyright)
install(FILES "${MYSWY_ROOT}/packaging/debian/README.Debian" DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/fcitx5-myswy")

if(BUILD_TESTING)
  add_executable(myswy-fcitx5-test test_engine.cpp)
  target_link_libraries(myswy-fcitx5-test PRIVATE myswy-engine)
  target_compile_options(myswy-fcitx5-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME fcitx5-engine COMMAND myswy-fcitx5-test)
  add_executable(myswy-reload-test test_dictionary_reload.cpp)
  target_link_libraries(myswy-reload-test PRIVATE myswy-engine)
  target_compile_options(myswy-reload-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME fcitx5-dictionary-reload COMMAND myswy-reload-test)
  set_tests_properties(fcitx5-engine fcitx5-dictionary-reload PROPERTIES TIMEOUT 15)
endif()
